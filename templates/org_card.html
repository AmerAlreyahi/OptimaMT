Quick start

import { pipeline } from "@huggingface/transformers";

const translator = await pipeline("translation", "$example", {
  dtype: "q8",
  session_options: { extra: { session: { disable_quant_qdq: "1" } } }, // fast mode
});
console.log((await translator("The lighthouse stands on the coast."))[0].translation_text);
